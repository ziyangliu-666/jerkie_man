# === 示例1: 横向排列 (T型结构) ===
# 三个房间水平排列，中间房间有四个门连接所有方向
@roomgroup layout=horizontal x=500 y=500 cellH=400
  west: w=300 doors=e
  center: w=500 doors=wens
  east: w=300 doors=w

# === 示例2: 纵向排列 ===
# 三个房间垂直堆叠，顶部房间有北门对外
@roomgroup layout=vertical x=100 y=600 cellW=300
  top: h=200 doors=ns
  mid: h=300 doors=ns
  bot: h=200 doors=n

# === 示例3: 2x2网格布局 ===
# 四个房间组成2x2网格，四个角有对外门
@roomgroup layout=grid x=1400 y=100 cols=2 rows=2 cellW=250 cellH=200
  nw: doors=nse
  ne: doors=nsw
  sw: doors=nse
  se: doors=nsw

# === 示例4: 复杂横向布局 ===
# 展示不同宽度的房间组合，两端有对外门
@roomgroup layout=horizontal x=200 y=200 cellH=250
  r1: w=200 doors=ne
  r2: w=400 doors=we
  r3: w=250 doors=we
  r4: w=200 doors=sw
